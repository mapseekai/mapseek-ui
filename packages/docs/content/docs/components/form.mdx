## 示例

<ShowcaseDemo registryName="form" demo="overview" title="Form 表单" description="组合字段、说明和校验错误来构建表单。" />

```tsx
import { useForm } from "react-hook-form"

import {
  Form,
  FormControl,
  FormField,
  FormItem,
  FormLabel,
  FormMessage,
} from "@/components/ui/form"
import { Input } from "@/components/ui/input"

const form = useForm({ defaultValues: { datasetName: "" } })

<Form {...form}>
  <form onSubmit={form.handleSubmit(console.log)}>
    <FormField
      control={form.control}
      name="datasetName"
      rules={{ required: "请输入数据集名称。" }}
      render={({ field }) => (
        <FormItem>
          <FormLabel>数据集名称</FormLabel>
          <FormControl>
            <Input {...field} />
          </FormControl>
          <FormMessage />
        </FormItem>
      )}
    />
  </form>
</Form>
```
